7장 — victim 선택과 direct victim

기준 소스: CUBRID 5cd4f860e · src/storage/page_buffer.c

이 장의 범위 새 페이지를 담을 프레임이 필요할 때 어느 BCB를 뽑아낼 것인가, 그리고 뽑을 것이 없을 때 어떻게 기다리고 어떻게 배급받는가를 규정한다. 6장이 “BCB가 어디에 놓이는가”였다면 이 장은 “BCB가 어떻게 회수되는가”다.

두 개의 독립된 메커니즘이 맞물린다: 능동 탐색(스레드가 직접 LRU 리스트를 훑는다)과 수동 배급(direct victim — 깨끗한 BCB를 만든 자가 대기 중인 스레드의 전용 슬롯에 꽂아 주고 깨운다). 정상 서버에서 압박이 걸리면 후자가 주된 공급원이 된다.

7.2.2 pgbuf_victimize_bcb page_buffer.c:8586-8631

/* 전제: 호출자가 bcb mutex 보유. 반환 시에도 계속 보유한다. */
int pgbuf_victimize_bcb (PGBUF_BCB *bufptr)
{
  /* 1) 재검증 — 탐색과 이 시점 사이에 상태가 바뀌었을 수 있다 */
  if (!pgbuf_is_bcb_victimizable (bufptr, /*has_mutex*/ true))
    { assert (false); PGBUF_BCB_UNLOCK (bufptr); return ER_FAILED; }

  /* 2) TO_VACUUM 플래그 해제 — 이 프레임은 더 이상 그 페이지가 아니다 */
  if (pgbuf_bcb_is_to_vacuum (bufptr))
    pgbuf_bcb_update_flags (bufptr, /*set*/ 0, PGBUF_BCB_TO_VACUUM_FLAG);

  assert (get_latch (&bufptr->atomic_latch) == PGBUF_NO_LATCH);

  /* 3) 해시 체인에서 제거 — 이제 아무도 이 VPID로 찾을 수 없다 */
  if (pgbuf_delete_from_hash_chain (bufptr) != NO_ERROR) return ER_FAILED;

  /* 4) latch 워드를 INVALID로 원자 저장 (fcnt=0, waiter_exists=false) */
  impl = get_impl (&bufptr->atomic_latch);
  impl.impl.latch_mode = PGBUF_LATCH_INVALID;
  impl.impl.waiter_exists = false;
  impl.impl.fcnt = 0;
  bufptr->atomic_latch.store (impl.raw);

  return NO_ERROR;                               /* mutex는 계속 보유 */
}

7.3 pgbuf_get_victim — 4단계 탐색 page_buffer.c:9019-9197

PGBUF_BCB *pgbuf_get_victim (void)
{
  detailed_perf    = perfmon_is_perf_tracking_and_active (PERFMON_ACTIVATION_FLAG_PB_VICTIMIZATION);
  has_flush_thread = pgbuf_is_page_flush_daemon_available ();
  restrict_other   = false;
  searched_own     = false;
  nloops           = 0;

  /* 요청 카운터: flush 데몬의 hit ratio 판정에 쓰이는 "miss" 측 값 */
  ATOMIC_INC_32 (&pgbuf_Pool.monitor.lru_victim_req_cnt, 1);

  /* ── 단계 1: 자기 private list — quota를 넘겼을 때만 ──────────── */
  if (PGBUF_THREAD_HAS_PRIVATE_LRU (thread_p))
    {
      private_lru_idx = PGBUF_LRU_INDEX_FROM_PRIVATE (PGBUF_PRIVATE_LRU_FROM_THREAD (thread_p));
      lru_list = PGBUF_GET_LRU_LIST (private_lru_idx);

      /* 두 조건 중 하나면 진입:
       *   (a) zone 1+2 만으로 이미 quota 초과, 또는
       *   (b) 리스트 전체가 quota 초과이면서 실제 후보가 존재 */
      if (PGBUF_LRU_LIST_IS_ONE_TWO_OVER_QUOTA (lru_list)
          || (PGBUF_LRU_LIST_IS_OVER_QUOTA (lru_list) && lru_list->count_vict_cand > 0))
        {
          victim = pgbuf_get_victim_from_lru_list (private_lru_idx);
          if (victim != NULL)
            {
              PERF (PSTAT_PB_OWN_VICTIM_PRIVATE_LRU_SUCCESS);
              PERF_TIME (PSTAT_PB_VICTIM_SEARCH_OWN_PRIVATE_LISTS);
              return victim;
            }
          PERF (PSTAT_PB_VICTIM_OWN_PRIVATE_LRU_FAIL);
          PERF_TIME (PSTAT_PB_VICTIM_SEARCH_OWN_PRIVATE_LISTS);

          /* 벌칙: quota를 크게 넘긴 스레드는 남의 private을 뒤지지 못한다.
           * 단 vacuum은 quota가 없으므로 제외.
           * "약간만" 넘긴 스레드(새 트랜잭션 등)는 봐준다 → 버퍼를 둔다. */
          if (!PGBUF_VACUUM_SHOULD_IGNORE_UNFIX (thread_p))
            restrict_other = PGBUF_LRU_LIST_IS_OVER_QUOTA_WITH_BUFFER (lru_list);
            /* 버퍼 = PGBUF_OVER_QUOTA_BUFFER(quota) = MAX (10, quota * 0.01) */

          searched_own = true;
        }
    }

  /* ── 단계 2: 남의 private list (lock-free 큐 경유) ─────────────── */
  if (PGBUF_PAGE_QUOTA_IS_ENABLED && has_flush_thread)
    {
      victim = pgbuf_lfcq_get_victim_from_private_lru (restrict_other);   /* §7.5 */
      if (victim != NULL) { PERF_TIME (PSTAT_PB_VICTIM_SEARCH_OTHERS_PRIVATE_LISTS); return victim; }
      PERF_TIME (PSTAT_PB_VICTIM_SEARCH_OTHERS_PRIVATE_LISTS);
    }

  /* ── 단계 3: shared list ───────────────────────────────────── */
  initial_cursor = pgbuf_Pool.shared_lrus_with_victims->get_consumer_cursor ();
  do
    {
      victim = pgbuf_lfcq_get_victim_from_shared_lru (has_flush_thread);
      if (victim != NULL) { PERF_TIME (PSTAT_PB_VICTIM_SEARCH_SHARED_LISTS); return victim; }
      current_cursor = pgbuf_Pool.shared_lrus_with_victims->get_consumer_cursor ();
    }
  while (!has_flush_thread                                   /* 멀티스레드면 1회만 */
         && !pgbuf_Pool.shared_lrus_with_victims->is_empty ()
         && ((int) (current_cursor - initial_cursor) <= pgbuf_Pool.num_LRU_list)
         && (++nloops <= pgbuf_Pool.num_LRU_list));         /* 무한 루프 안전장치 */
  PERF_TIME (PSTAT_PB_VICTIM_SEARCH_SHARED_LISTS);

  PERF (PSTAT_PB_VICTIM_ALL_LRU_FAIL);

  /* ── 단계 4: 최후 — 자기 private을 quota와 무관하게 다시 뒤진다 ── */
  if (PGBUF_THREAD_HAS_PRIVATE_LRU (thread_p) && !searched_own)
    {
      private_lru_idx = PGBUF_LRU_INDEX_FROM_PRIVATE (PGBUF_PRIVATE_LRU_FROM_THREAD (thread_p));
      victim = pgbuf_get_victim_from_lru_list (private_lru_idx);
      if (victim != NULL) { PERF (PSTAT_PB_OWN_VICTIM_PRIVATE_LRU_SUCCESS); return victim; }
      PERF (PSTAT_PB_VICTIM_OWN_PRIVATE_LRU_FAIL);
    }

  return NULL;
}
단계 4가 존재하는 이유 (소스 주석 9054-9056) “위의 모든 시도가 실패하면, quota 미만이더라도 자기 private에서 뽑아 본다. 모든 private 리스트가 quota에 거의 맞닿아 있고(그러나 조금 미달), shared 리스트에는 zone 3이 아예 없어서 아무것도 교체하거나 flush할 수 없는 이상한 사례를 발견했다.” — 즉 이 단계는 이론적 완결성이 아니라 실제로 관측된 교착 상황에 대한 대응이다. MUST 재현해야 한다.
결함 D4 — big private 큐에는 생산자가 없다 단계 2에서 restrict_other == true인 스레드는 pgbuf_lfcq_get_victim_from_private_lru의 big 큐만 볼 수 있다 page_buffer.c:16382-16385. 그런데 big_private_lrus_with_victims에 produce하는 지점은 파일 전체에 단 한 곳, 같은 함수 안에서 big 큐 consume에 성공한 뒤의 재등록 경로뿐이다 page_buffer.c:16401. 생산자 pgbuf_lfcq_add_lru_with_victims는 private_lrus_with_victims와 shared_lrus_with_victims에만 넣는다 16329, 16337.

귀결: big 큐는 부팅부터 종료까지 영구히 비어 있다. PSTAT_PB_LFCQ_LRU_PRV_GET_BIG은 절대 증가하지 않고, restrict_other가 켜진 스레드는 단계 2에서 항상 즉시 NULL을 받아 곧장 shared 리스트로 간다. 즉 “quota 초과 스레드를 남의 private에서 배제한다”는 벌칙은 의도한 “큰 리스트에서는 허용” 완화 없이 전면 배제로 작동한다.

선택지 — 충실 재현(권장) 코드를 그대로 옮긴다. big 큐를 할당은 하되 (PGBUF_PRIVATE_LRU_COUNT * 2 1827) 아무도 채우지 않는 상태를 유지한다. 관측 카운터가 원본과 일치한다. 수정 pgbuf_lfcq_add_lru_with_victims에 “리스트가 크고 quota를 크게 넘겼으면 big 큐로 라우팅” 분기를 추가한다(§7.5의 재등록 조건과 같은 임계를 쓰면 자연스럽다). victim 분포와 Num_lfcq_* 계열이 모두 달라지므로 1차 구현에서는 하지 않는다.

    { PERF (..._WAS_EMPTY); pthread_mutex_unlock (...); return NULL; }

  /* ── 5) hint 스냅: 바닥이 깨끗한데 hint가 딴 데를 가리키면 바닥으로 되돌린다 ── */
  if (!pgbuf_bcb_is_dirty (lru_list->bottom) && lru_list->victim_hint != lru_list->bottom)
    ATOMIC_TAS_ADDR (&lru_list->victim_hint, lru_list->bottom);

  /* ── 6) hint(없으면 bottom)에서 위로 스캔 ─────────────────── */
  found_victim_cnt = 0; bufptr_victimizable = NULL;
  victim_hint  = lru_list->victim_hint;
  bufptr_start = (victim_hint != NULL) ? victim_hint : lru_list->bottom;

  for (bufptr = bufptr_start, search_cnt = 0;
       bufptr != NULL && PGBUF_IS_BCB_IN_LRU_VICTIM_ZONE (bufptr) && search_cnt < MAX_DEPTH;
       bufptr = bufptr->prev_BCB, search_cnt++)
    {
      /* 6a) dirty / flushing / 이미 direct victim ⇒ 후보 자격 자체가 없다 */
      if (pgbuf_bcb_avoid_victim (bufptr))
        continue;

      /* 6b) 누군가 fix 중이거나 대기자가 있다 ⇒ "유효한 후보지만 지금은 못 쓴다" */
      if (pgbuf_is_bcb_fixed_by_any (bufptr, /*has_mutex*/ false))
        {
          if (bufptr_victimizable == NULL)
            {
              bufptr_victimizable = bufptr;
              /* 첫 번째 그런 BCB로 hint를 전진시킨다 (그 사이 hint가 안 바뀌었을 때만) */
              if (bufptr_victimizable != victim_hint)
                ATOMIC_CAS_ADDR (&lru_list->victim_hint, victim_hint, bufptr_victimizable);
            }
          if (++found_victim_cnt >= lru_victim_cnt)
            break;                       /* 조기 종료: 후보 수만큼 봤으면 더 없을 것이다 */
          continue;
        }

      /* 6c) 진짜 후보다. 리스트 mutex를 쥔 채이므로 BCB 락은 "조건부"로만 */
      if (PGBUF_BCB_TRYLOCK (bufptr) == 0)
        {
          if (pgbuf_is_bcb_victimizable (bufptr, /*has_mutex*/ true))
            {
              if (bufptr_victimizable == NULL)
                pgbuf_lru_advance_victim_hint (lru_list, victim_hint, bufptr->prev_BCB, false);

              pgbuf_remove_from_lru_list (bufptr, lru_list);    /* zone := VOID */

#if defined (SERVER_MODE)
              /* todo: this is a hack — low 대기자가 많으면 여기서 무더기 배급을 시도 */
              if (waiter_threads_low_priority->size () >= (5 + thread_num_total_threads () / 20))
                pgbuf_panic_assign_direct_victims_from_lru (lru_list, bufptr->prev_BCB);
#endif
              /* 새 바닥이 dirty면 flush 데몬을 깨워 다음 후보를 만들게 한다 */
              if (lru_list->bottom != NULL && pgbuf_bcb_is_dirty (lru_list->bottom)
                  && pgbuf_is_page_flush_daemon_available ())
                pgbuf_wakeup_page_flush_daemon ();

              pthread_mutex_unlock (&lru_list->mutex);         /* 리스트 락 해제 */
              pgbuf_add_vpid_to_aout_list (&bufptr->vpid, lru_idx);
              return bufptr;                                    /* BCB는 "잠긴 채" 반환 */
            }
          PGBUF_BCB_UNLOCK (bufptr);
        }
      else
        {
          /* trylock 실패: 단일 스레드 환경에서는 불가능해야 한다 */
          assert (pgbuf_is_page_flush_daemon_available ());
          if (bufptr_victimizable == NULL)
            {
              bufptr_victimizable = bufptr;
              if (bufptr != victim_hint)
                ATOMIC_CAS_ADDR (&lru_list->victim_hint, victim_hint, bufptr_victimizable);
            }
          if (++found_victim_cnt >= lru_victim_cnt)
            break;
        }
    }

  /* ── 7) 전체 실패 ─────────────────────────────────────── */
  PERF (PSTAT_PB_VICTIM_GET_FROM_LRU_FAIL);
  if (bufptr_victimizable == NULL && victim_hint != NULL)
    {
      /* hint가 있었는데 후보를 하나도 못 봤다 ⇒ hint가 틀렸다 */
      PERF (PSTAT_PB_VICTIM_GET_FROM_LRU_BAD_HINT);
      if (lru_list->count_vict_cand > 0 && PGBUF_IS_BCB_IN_LRU_VICTIM_ZONE (lru_list->bottom))
        ATOMIC_CAS_ADDR (&lru_list->victim_hint, victim_hint, lru_list->bottom);
      else
        ATOMIC_CAS_ADDR (&lru_list->victim_hint, victim_hint, (PGBUF_BCB *) NULL);
    }

  pthread_mutex_unlock (&lru_list->mutex);
  pgbuf_wakeup_page_flush_daemon ();          /* 후보가 더 필요하다 */
  return NULL;
}

PGBUF_BCB *pgbuf_get_direct_victim (void)
{
  /* 슬롯을 원자적으로 비우며 가져온다 */
  bcb = ATOMIC_TAS_ADDR (&pgbuf_Pool.direct_victims.bcb_victims[thread_p->index], (PGBUF_BCB *) NULL);
  assert (bcb != NULL);

  PGBUF_BCB_LOCK (bcb);                             /* ← BCB 락 */

  /* 배급 이후 누군가 이 페이지를 다시 fix했다면 무효 표시가 서 있다 */
  if (pgbuf_bcb_is_invalid_direct_victim (bcb))
    {
      pgbuf_bcb_update_flags (bcb, 0, PGBUF_BCB_INVALIDATE_DIRECT_VICTIM_FLAG);  /* 표시 해제 */
      PGBUF_BCB_UNLOCK (bcb);
      return NULL;                                  /* 호출자는 high priority로 재파킹 */
    }

  assert (pgbuf_bcb_is_direct_victim (bcb));
  pgbuf_bcb_update_flags (bcb, 0, PGBUF_BCB_VICTIM_DIRECT_FLAG);   /* 소유권 인수 */

  if (!pgbuf_is_bcb_victimizable (bcb, true))
    { assert (false); PGBUF_BCB_UNLOCK (bcb); return NULL; }       /* 일어나면 안 된다 */

  switch (pgbuf_bcb_get_zone (bcb))
    {
    case PGBUF_VOID_ZONE:   break;                  /* 이미 리스트 밖 */
    case PGBUF_INVALID_ZONE: assert (false); break;
    default:                                        /* 아직 LRU에 남아 있다 */
      lru_idx = pgbuf_bcb_get_lru_index (bcb);
      pgbuf_lru_remove_bcb (bcb);                   /* 리스트 mutex 획득/해제, zone := VOID */
      pgbuf_add_vpid_to_aout_list (&bcb->vpid, lru_idx);
      break;
    }

  assert (pgbuf_bcb_get_zone (bcb) == PGBUF_VOID_ZONE);
  return bcb;                                       /* mutex는 잠긴 채 반환 */
}

7.6.5 재fix 무효화 규약

규범 N7-4 — 배급된 BCB를 다시 fix하는 경우 배급된 BCB는 아직 해시 체인에 살아 있으므로, 대기 스레드가 깨어나기 전에 다른 워커가 그 VPID를 fix할 수 있다 page_buffer.c:228-233.
MUST fix 경로는 PGBUF_BCB_VICTIM_DIRECT_FLAG가 서 있는 BCB를 집으면 같은 update 연산으로 VICTIM_DIRECT를 내리고 INVALIDATE_DIRECT_VICTIM을 세운다 — 예: pgbuf_simple_fix page_buffer.c:2706-2710 (5장의 fix 결정 트리 참조).
MUST 대기 스레드는 §7.6.4에서 이를 발견하고 NULL을 반환하며, pgbuf_allocate_bcb는 high priority로 승격해 재파킹한다 page_buffer.c:8265-8270. 승격이 없으면 재fix가 반복될 때 그 스레드가 굶는다.
MUST fix가 필요하지 않은 조회 경로(need_fix == false)는 direct victim 표시가 선 BCB를 “찾지 못한 것”으로 취급해 NULL을 반환한다 page_buffer.c:2714-2719 — 곧 해시에서 사라질 프레임이기 때문이다.

/* page_buffer.c:9493-9549. 주석 9501: "statistics shows not useful" */
int pgbuf_panic_assign_direct_victims_from_lru (PGBUF_LRU_LIST *lru_list, PGBUF_BCB *bcb_start)
{
#define MAX_DEPTH 1000
  if (bcb_start == NULL) return 0;
  assert (pgbuf_bcb_get_lru_index (bcb_start) == lru_list->index);

  for (bcb = bcb_start, count = 0, n_assigned = 0;
       bcb != NULL && PGBUF_IS_BCB_IN_LRU_VICTIM_ZONE (bcb)
       && lru_list->count_vict_cand > 0 && count < MAX_DEPTH;
       bcb = bcb->prev_BCB, count++)
    {
      if (!pgbuf_is_bcb_victimizable (bcb, false)) continue;
      if (PGBUF_BCB_TRYLOCK (bcb) != 0)           continue;   /* 조건부만 */
      if (!pgbuf_is_bcb_victimizable (bcb, true)) { PGBUF_BCB_UNLOCK (bcb); continue; }

      if (!pgbuf_assign_direct_victim (bcb))
        { PGBUF_BCB_UNLOCK (bcb); break; }        /* 대기자가 더 없다 ⇒ 즉시 종료 */

      PGBUF_BCB_UNLOCK (bcb);
      perfmon_inc (PSTAT_PB_VICTIM_ASSIGN_DIRECT_PANIC);      /* 상세 통계 시 */
      n_assigned++;
    }
  return n_assigned;
#undef MAX_DEPTH
}

/* page_buffer.c:9602-9635 */
void pgbuf_lfcq_assign_direct_victims (int lru_idx, int *nassign_inout)
{
  lru_list = PGBUF_GET_LRU_LIST (lru_idx);
  if (lru_list->count_vict_cand > 0)
    {
      pthread_mutex_lock (&lru_list->mutex);
      victim_hint = lru_list->victim_hint;
      nassigned = pgbuf_panic_assign_direct_victims_from_lru (lru_list, victim_hint);

      if (nassigned == 0 && lru_list->count_vict_cand > 0
          && pgbuf_is_any_thread_waiting_for_direct_victim ())
        {
          /* hint가 나빴을 가능성이 가장 크다 ⇒ 바닥으로 리셋하고 다시 */
          if (PGBUF_IS_BCB_IN_LRU_VICTIM_ZONE (lru_list->bottom))
            ATOMIC_CAS_ADDR (&lru_list->victim_hint, victim_hint, lru_list->bottom);
          else
            ATOMIC_CAS_ADDR (&lru_list->victim_hint, victim_hint, (PGBUF_BCB *) NULL);

          nassigned = pgbuf_panic_assign_direct_victims_from_lru (lru_list, lru_list->bottom);
        }
      pthread_mutex_unlock (&lru_list->mutex);
      (*nassign_inout) -= nassigned;
    }
}

7.7 결함 D3 — pgbuf_direct_victims_maintenance는 no-op이다

결함 D3 — 루프 조건이 진입 시점에 거짓 page_buffer.c:9560-9592. 이 함수는 “시스템 활동이 거의 없어 아무도 대기자를 먹여 주지 못하는 상황”을 위한 백업 플랜으로 설계되었다(주석 9551-9556). 그런데 두 루프가 모두 이렇게 쓰여 있다: 
static int prv_index = 0;
for (index = prv_index, restarted = false;
     pgbuf_is_any_thread_waiting_for_direct_victim () && nassigns > 0
     && index != prv_index && !restarted;                    /* ← index == prv_index로 시작 */
     (index == PGBUF_PRIVATE_LRU_COUNT - 1) ? index = 0, restarted = true : index++)
  {
    pgbuf_lfcq_assign_direct_victims (PGBUF_LRU_INDEX_FROM_PRIVATE (index), &nassigns);
  }
prv_index = index;
index가 prv_index로 초기화되므로 index != prv_index는 첫 평가에서 항상 거짓이다. 두 루프 모두 몸통이 한 번도 실행되지 않으며, prv_index/shr_index도 값이 바뀌지 않는다. 결과적으로 100 ms 유지보수 데몬의 후반부 page_buffer.c:16942는 아무 일도 하지 않는다.

선택지 — 충실 재현(권장) 함수를 그대로 옮기고 “이 루프는 실행되지 않는다”는 주석을 단다. 동작이 원본과 동일하고, 대기자 급전은 §7.6.2의 배급자 6종과 pgbuf_allocate_bcb의 flush 데몬 기상으로 충분히 공급된다(quiz 09가 이를 실증한다). 수정 조건을 라운드로빈 의도대로 고친다(예: restarted만으로 종료를 판정하고 index != prv_index를 제거). 그러면 유휴 시스템에서 100 ms마다 최대 5개의 BCB가 추가로 배급되고 Num_victim_assign_direct_panic이 새로 관측된다 — 동작이 바뀌므로 1차 구현에서는 하지 않는다.

UNVERIFIED: 이것이 의도된 비활성인지 오타인지는 소스만으로 판정할 수 없다. 감싸는 주석은 이 루프가 실행되어야 한다고 서술한다.

7.10 수용 기준

수용 기준 A7 — victim/direct victim (마일스톤 M4) quiz 09 (victim under pressure): 풀보다 큰 데이터에 UPDATE를 걸어 “모든 후보가 dirty”인 상태를 만들었을 때 —

  3. Num_alloc_bcb_wait_threads_low_priority 게이지가 압박 구간에서 양수로 관측되고, alloc_bcb_cond_wait_low_prio 시간이 쌓여야 한다 — 파킹이 실제로 일어났다는 뜻.
  4. Num_log_wals > 0 — victim flush가 WAL flush를 끌어냈다(4장·8장의 WAL 규칙이 여기서 대량 발동한다).
  5. UPDATE 종료 직후에도 Num_data_page_dirty 게이지가 높게 남아야 한다 — flush 데몬은 victim 수요만큼만 흘려보내고 나머지는 checkpoint 몫이다.
